## Live Telemetry Example

Here is the helper function this portfolio uses to fetch real-time analytics from Umami Cloud with a 60-second stale-while-revalidate cache:

```typescript
export async function getLiveMetrics(websiteId: string) {
  const res = await fetch(`https://api.umami.is/v1/websites/${websiteId}/stats`, {
    next: { revalidate: 60 },
  });
  
  if (!res.ok) {
    throw new Error(`Failed to fetch metrics: ${res.statusText}`);
  }
  
  return res.json();
}
```
